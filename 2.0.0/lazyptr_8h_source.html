28template <typename T>
29class LazyPtr
30{
31private:
32 std::atomic<T*> m_ptr{nullptr};
33
34public: // ...structors
35 LazyPtr() = default;
36
37 // Clean up allocated resource on destruction
38 ~LazyPtr()
39 {
40 delete m_ptr.load(std::memory_order_relaxed);
41 }
42
43 // Prevent copying to avoid double-free errors
44 LazyPtr(const LazyPtr&) = delete;
45 LazyPtr& operator=(const LazyPtr&) = delete;
46
47 // Allow moving
48 LazyPtr(LazyPtr&& other) noexcept
49 : m_ptr(other.m_ptr.exchange(nullptr, std::memory_order_relaxed))
50 {}
51
52public: // Operators
53 // Allow assignment
54 LazyPtr& operator=(LazyPtr&& other) noexcept
55 {
56 if (this != &other)
57 {
58 delete m_ptr.load(std::memory_order_relaxed);
59 m_ptr.store(other.m_ptr.exchange(nullptr, std::memory_order_relaxed),
60 std::memory_order_relaxed);
61 }
62 return *this;
63 }
64
65 // Lazy initialization & dereference
66 T& operator*() { return *get_or_throw(); }
67
68 // Lazy initialization
69 T* operator->() { return get_or_throw(); }
70
71 // Explicit bool conversion
72 explicit operator bool() const noexcept
73 {
74 return m_ptr.load(std::memory_order_relaxed) != nullptr;
75 }
76
77 // The raw pointer value
78 T* get_ptr() const noexcept
79 {
80 return m_ptr.load(std::memory_order_relaxed);
81 }
82
83private: // Modifiers
84 // Lazily initialize if required
85 T* get_or_throw()
86 {
87 T* p = m_ptr.load(std::memory_order_relaxed);
88 if (!p)
89 {
90 auto* new_p = new T();
91 if (!m_ptr.compare_exchange_strong(p, new_p
92 , std::memory_order_relaxed
93 , std::memory_order_relaxed
94 )
95 )
96 {
97 delete new_p; // Lost the race
98 }
99 else
100 {
101 p = new_p;
102 }
103 }
104 return p;
105 }
106};
